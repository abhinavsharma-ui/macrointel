// ── Breakdown cards ──────────────────────────────────────────────
function fillBreakdown(d) {
  const a = d.analytics || {};
  const pos = (d.portfolio || {}).positions || [];

  // ── Closed Outcomes ──
  const outBadge = document.getElementById("outBadge");
  const outDiv   = document.getElementById("outBreakdown");
  const n = a.closed_trades || 0;
  if (outBadge) outBadge.textContent = n + " trades";
  if (outDiv) {
    if (n > 0) {
      const wr   = Number(a.win_rate || 0);
      const wins = Math.round(wr * n / 100);
      const loss = n - wins;
      const pt   = a.profit_target_hit_rate != null ? Number(a.profit_target_hit_rate).toFixed(1) : "--";
      const targetPct = 60.6;
      outDiv.innerHTML = `
        <div style="display:grid;grid-template-columns:1fr 1fr 1fr;gap:8px;margin-bottom:14px">
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">&#9650; Wins</div>
            <div style="font:700 22px Consolas;color:var(--g)">${wins}</div>
          </div>
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">&#9660; Losses</div>
            <div style="font:700 22px Consolas;color:var(--r)">${loss}</div>
          </div>
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">PT Hit</div>
            <div style="font:700 22px Consolas;color:var(--t)">${pt}%</div>
          </div>
        </div>
        <div style="border-top:1px solid var(--l);padding-top:12px">
          <div style="display:flex;justify-content:space-between;margin-bottom:6px">
            <span class="label">Win rate</span>
            <span style="font:11px Consolas;color:var(--mut)">${wr.toFixed(1)}% <span style="color:var(--lo)">vs ${targetPct}% target</span></span>
          </div>
          <div class="bar">
            <div class="fill green" style="width:${Math.min(wr,100)}%;position:relative">
              <div style="position:absolute;right:${Math.max(0,(wr-targetPct)/wr*100).toFixed(1)}%;top:-2px;bottom:-2px;width:2px;background:var(--a);opacity:.8;border-radius:1px"></div>
            </div>
          </div>
          <div style="display:flex;justify-content:space-between;margin-top:4px;font-size:10px;color:var(--lo)">
            <span>0%</span><span style="color:var(--a)">&#9650; ${targetPct}% target</span><span>100%</span>
          </div>
        </div>`;
    } else {
      outDiv.innerHTML = '<div class="empty">No closed trades yet</div>';
    }
  }

  // ── Unrealized P&L Breakdown ──
  const obBadge = document.getElementById("ob");
  const posDiv  = document.getElementById("posBreakdown");
  if (obBadge) obBadge.textContent = pos.length + " open";
  if (posDiv) {
    if (pos.length > 0) {
      const winners = pos.filter(x => Number(x.unrealized_pnl||0) > 0);
      const losers  = pos.filter(x => Number(x.unrealized_pnl||0) < 0);
      const flat    = pos.filter(x => Number(x.unrealized_pnl||0) === 0);
      const total   = pos.reduce((s,x) => s + Number(x.unrealized_pnl||0), 0);
      const winPct  = (winners.length / pos.length * 100).toFixed(0);
      const sign    = total >= 0 ? "+" : "";
      const col     = total >= 0 ? "var(--g)" : "var(--r)";
      posDiv.innerHTML = `
        <div style="display:grid;grid-template-columns:1fr 1fr 1fr;gap:8px;margin-bottom:14px">
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">&#9650; Up</div>
            <div style="font:700 22px Consolas;color:var(--g)">${winners.length}</div>
          </div>
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">&#9660; Down</div>
            <div style="font:700 22px Consolas;color:var(--r)">${losers.length}</div>
          </div>
          <div style="background:var(--p2);border-radius:6px;padding:10px 8px;text-align:center">
            <div class="label" style="margin-bottom:5px">Flat</div>
            <div style="font:700 22px Consolas;color:var(--mut)">${flat.length}</div>
          </div>
        </div>
        <div style="border-top:1px solid var(--l);padding-top:12px">
          <div style="display:flex;justify-content:space-between;margin-bottom:6px">
            <span class="label">Winners vs losers</span>
            <span style="font:11px Consolas;color:var(--mut)">${winners.length} / ${pos.length}</span>
          </div>
          <div class="bar">
            <div class="fill green" style="width:${winPct}%"></div>
          </div>
          <div style="display:flex;justify-content:space-between;align-items:center;margin-top:10px">
            <span class="label">Total unrealized P&L</span>
            <span style="font:700 18px Consolas;color:${col}">${sign}$${Math.abs(total).toFixed(2)}</span>
          </div>
        </div>`;
    } else {
      posDiv.innerHTML = '<div class="empty">No open positions</div>';
    }
  }
}
